# Data Table Design Document
## TacoLearn - Database Schema and Data Model

---

## 1. Core Domain Models

### User Management

#### users
```sql
CREATE TABLE users (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  email VARCHAR(255) UNIQUE NOT NULL,
  password_hash VARCHAR(255) NOT NULL,
  full_name VARCHAR(255) NOT NULL,
  avatar_url TEXT,
  role ENUM ('STUDENT', 'TEACHER', 'ADMIN') NOT NULL,
  is_active BOOLEAN DEFAULT true,
  last_login_at TIMESTAMP,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX idx_users_email ON users(email);
CREATE INDEX idx_users_role ON users(role);
```

#### students
```sql
CREATE TABLE students (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID UNIQUE NOT NULL REFERENCES users(id) ON DELETE CASCADE,
  jlpt_goal_level ENUM ('N1', 'N2', 'N3', 'N4', 'N5') NOT NULL,
  jlpt_goal_date DATE,
  current_readiness_score DECIMAL(5,2) DEFAULT 0,
  total_practice_time_hours DECIMAL(10,2) DEFAULT 0,
  bio TEXT,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
```

#### teachers
```sql
CREATE TABLE teachers (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID UNIQUE NOT NULL REFERENCES users(id) ON DELETE CASCADE,
  specialization VARCHAR(255),
  bio TEXT,
  years_of_experience INTEGER,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
```

### Learning Content

#### questions
```sql
CREATE TABLE questions (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  jlpt_level ENUM ('N1', 'N2', 'N3', 'N4', 'N5') NOT NULL,
  section VARCHAR(100) NOT NULL,  -- Vocabulary, Grammar, Kanji, Reading, Listening
  skill VARCHAR(200) NOT NULL,     -- Author's Opinion, Inference, Main Idea, etc.
  question_type VARCHAR(100),      -- Multiple Choice, Fill-in, Matching
  difficulty ENUM ('EASY', 'MEDIUM', 'HARD') NOT NULL,
  
  question_content TEXT NOT NULL,
  question_image_url TEXT,
  
  choices JSONB NOT NULL,          -- [{text, code}, ...]
  correct_answer_code VARCHAR(10) NOT NULL,
  explanation TEXT NOT NULL,
  
  related_vocabulary UUID[],       -- Array of vocabulary IDs
  related_grammar UUID[],          -- Array of grammar IDs
  tags TEXT[],                     -- Array of tags
  
  difficulty_index DECIMAL(5,2),   -- Computed difficulty based on attempts
  attempt_count INTEGER DEFAULT 0,
  correct_count INTEGER DEFAULT 0,
  
  source VARCHAR(100),             -- Official, Community, AI-Generated
  created_by UUID REFERENCES users(id),
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX idx_questions_level_section_skill 
  ON questions(jlpt_level, section, skill);
CREATE INDEX idx_questions_difficulty ON questions(difficulty);
```

#### question_attempts
```sql
CREATE TABLE question_attempts (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  student_id UUID NOT NULL REFERENCES students(id) ON DELETE CASCADE,
  question_id UUID NOT NULL REFERENCES questions(id),
  session_id UUID,                 -- Links to practice_session
  
  student_answer_code VARCHAR(10) NOT NULL,
  is_correct BOOLEAN NOT NULL,
  time_spent_seconds INTEGER,
  
  confidence_level ENUM ('VERY_LOW', 'LOW', 'MEDIUM', 'HIGH', 'VERY_HIGH'),
  hint_used BOOLEAN DEFAULT false,
  marked_for_review BOOLEAN DEFAULT false,
  
  mistake_type ENUM ('CARELESS', 'MISUNDERSTANDING', 'SKILL_GAP', 'TIME_PRESSURE'),
  
  attempted_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX idx_question_attempts_student_date 
  ON question_attempts(student_id, attempted_at);
CREATE INDEX idx_question_attempts_question 
  ON question_attempts(question_id);
```

### Skill Mastery & Learning Progress

#### skill_mastery
```sql
CREATE TABLE skill_mastery (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  student_id UUID NOT NULL REFERENCES students(id) ON DELETE CASCADE,
  jlpt_level ENUM ('N1', 'N2', 'N3', 'N4', 'N5') NOT NULL,
  section VARCHAR(100),             -- Top-level: Vocabulary, Grammar, Reading, etc.
  skill VARCHAR(200) NOT NULL,      -- Sub-skill: Author's Opinion, Inference, etc.
  
  correct_answers INTEGER DEFAULT 0,
  total_attempts INTEGER DEFAULT 0,
  mastery_percentage DECIMAL(5,2) DEFAULT 0,  -- Calculated: (correct / total) * 100
  
  -- Spaced Repetition tracking
  last_attempted_at TIMESTAMP,
  first_attempted_at TIMESTAMP,
  
  -- Trend tracking
  weekly_progress DECIMAL(5,2),     -- Change in mastery % this week
  trend_direction ENUM ('IMPROVING', 'STABLE', 'DECLINING'),
  
  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  
  UNIQUE(student_id, jlpt_level, skill)
);

CREATE INDEX idx_skill_mastery_student_level 
  ON skill_mastery(student_id, jlpt_level);
```

#### daily_learning_plans
```sql
CREATE TABLE daily_learning_plans (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  student_id UUID NOT NULL REFERENCES students(id) ON DELETE CASCADE,
  plan_date DATE NOT NULL,
  
  activities JSONB NOT NULL,  -- [{type, skill, count, time_minutes, priority}, ...]
  total_time_minutes INTEGER,
  
  status ENUM ('PENDING', 'IN_PROGRESS', 'COMPLETED', 'PARTIAL') DEFAULT 'PENDING',
  started_at TIMESTAMP,
  completed_at TIMESTAMP,
  
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  
  UNIQUE(student_id, plan_date)
);
```

### Spaced Repetition System

#### srs_items
```sql
CREATE TABLE srs_items (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  student_id UUID NOT NULL REFERENCES students(id) ON DELETE CASCADE,
  content_id UUID NOT NULL,         -- question_id, vocabulary_id, etc.
  content_type ENUM ('QUESTION', 'VOCABULARY', 'GRAMMAR', 'KANJI'),
  
  -- Spaced Repetition intervals (SM-2 algorithm)
  stage INTEGER DEFAULT 0,          -- 0-5 (graduation = no review)
  ease_factor DECIMAL(4,2) DEFAULT 2.5,
  interval_days INTEGER DEFAULT 1,
  
  -- Review tracking
  review_count INTEGER DEFAULT 0,
  last_reviewed_at TIMESTAMP,
  next_review_at TIMESTAMP,
  
  -- Confidence tracking
  last_confidence_rating ENUM ('VERY_LOW', 'LOW', 'MEDIUM', 'HIGH', 'VERY_HIGH'),
  average_confidence DECIMAL(3,2),
  
  -- Status
  status ENUM ('NEW', 'LEARNING', 'REVIEW', 'GRADUATED') DEFAULT 'NEW',
  
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX idx_srs_items_next_review 
  ON srs_items(student_id, next_review_at) 
  WHERE status != 'GRADUATED';
```

### School & Class Management

#### schools
```sql
CREATE TABLE schools (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  name VARCHAR(255) NOT NULL,
  owner_id UUID NOT NULL REFERENCES users(id),
  description TEXT,
  logo_url TEXT,
  address TEXT,
  city VARCHAR(100),
  country VARCHAR(100),
  subscription_plan ENUM ('BASIC', 'PRO', 'ENTERPRISE'),
  subscription_status ENUM ('ACTIVE', 'TRIAL', 'SUSPENDED', 'CANCELLED'),
  max_students INTEGER,
  is_active BOOLEAN DEFAULT true,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
```

#### classes
```sql
CREATE TABLE classes (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  school_id UUID NOT NULL REFERENCES schools(id) ON DELETE CASCADE,
  teacher_id UUID NOT NULL REFERENCES users(id),
  name VARCHAR(255) NOT NULL,
  jlpt_level ENUM ('N1', 'N2', 'N3', 'N4', 'N5') NOT NULL,
  schedule JSONB,                   -- [{day, time, duration}, ...]
  max_students INTEGER DEFAULT 30,
  status ENUM ('ACTIVE', 'ARCHIVED') DEFAULT 'ACTIVE',
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
```

#### class_enrollments
```sql
CREATE TABLE class_enrollments (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  class_id UUID NOT NULL REFERENCES classes(id) ON DELETE CASCADE,
  student_id UUID NOT NULL REFERENCES students(id) ON DELETE CASCADE,
  enrolled_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  dropped_at TIMESTAMP,
  status ENUM ('ACTIVE', 'DROPPED') DEFAULT 'ACTIVE',
  
  UNIQUE(class_id, student_id)
);
```

#### attendance
```sql
CREATE TABLE attendance (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  class_id UUID NOT NULL REFERENCES classes(id) ON DELETE CASCADE,
  student_id UUID NOT NULL REFERENCES students(id) ON DELETE CASCADE,
  attendance_date DATE NOT NULL,
  
  status ENUM ('PRESENT', 'ABSENT', 'LATE', 'EXCUSED') NOT NULL,
  notes TEXT,
  
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  
  UNIQUE(class_id, student_id, attendance_date)
);
```

### Announcements & Notifications

## 2. Data Relationships Diagram

```
Users (1)
  ├── (1) Students
  │    ├── (N) QuestionAttempts
  │    ├── (N) SkillMastery
  │    ├── (N) SRSItems
  │    ├── (N) DailyLearningPlans
  │    ├── (N) PracticeSessions
  │    └── (N) ClassEnrollments

  │              └── Teachers (User)
  ├── (1) Teachers
  │    └── (N) Classes
  └── (1) AdminUser

Questions (1)
  ├── (N) QuestionAttempts
  ├── (N) SRSItems
  └── (N) PracticeSessions
```

## 4. Indexes for Performance

Key indexes for common queries:
- `idx_question_attempts_student_date` - Find recent attempts
- `idx_skill_mastery_student_level` - Mastery lookup
- `idx_srs_items_next_review` - Find items due for review
- `idx_questions_level_section_skill` - Question filtering
