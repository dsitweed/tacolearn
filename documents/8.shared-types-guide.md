## 1. User Types

### Enums
```typescript
// src/types/user.types.ts

export enum UserRole {
  STUDENT = 'STUDENT',
  TEACHER = 'TEACHER',
  ADMIN = 'ADMIN'
}

export enum UserStatus {
  ACTIVE = 'ACTIVE',
  INACTIVE = 'INACTIVE',
  SUSPENDED = 'SUSPENDED'
}
```

### User Type
```typescript
export interface User {
  id: string;
  email: string;
  fullName: string;
  avatarUrl?: string;
  role: UserRole;
  status: UserStatus;
  lastLoginAt?: Date;
  createdAt: Date;
  updatedAt: Date;
}

export type StudentUser = User & { role: UserRole.STUDENT };
export type TeacherUser = User & { role: UserRole.TEACHER };
export type AdminUser = User & { role: UserRole.ADMIN };
```

```typescript
export class AuthResponseDto {

```

### Practice DTOs
```typescript
// src/practice/dto/create-practice-session.dto.ts
export class CreatePracticeSessionDto {
  @IsEnum(['ADAPTIVE', 'TARGETED', 'MIXED', 'WEAK_AREAS'])
  type: string;
  
  @IsEnum(JLPTLevel)
  level: JLPTLevel;
  
  @IsOptional()
  skill?: string;
  
  @IsInt()
  @Min(1)
  @Max(50)
  questionCount: number;
}

// src/practice/dto/submit-answer.dto.ts
export class SubmitAnswerDto {
  @IsUUID()
  questionId: string;
  
  @IsIn(['A', 'B', 'C', 'D'])
  answer: string;
  
  @IsInt()
  @Min(0)
  timeSpent: number;
  
  @IsOptional()
  @IsEnum(['VERY_LOW', 'LOW', 'MEDIUM', 'HIGH', 'VERY_HIGH'])
  confidence?: string;
}
```

## 10. Type Utility Functions

```typescript
// src/utils/type-guards.ts

export function isStudent(user: User): user is StudentUser {
  return user.role === UserRole.STUDENT;
}

export function isTeacher(user: User): user is TeacherUser {
  return user.role === UserRole.TEACHER;
}

export function isAdmin(user: User): user is AdminUser {
  return user.role === UserRole.ADMIN;
}

export function isMastered(mastery: number): boolean {
  return mastery >= 80;
}

export function isWeak(mastery: number): boolean {
  return mastery < 50;
}

export function getMasteryLevel(mastery: number): 'EXCELLENT' | 'GOOD' | 'WARNING' | 'WEAK' {
  if (mastery >= 80) return 'EXCELLENT';
  if (mastery >= 65) return 'GOOD';
  if (mastery >= 50) return 'WARNING';
  return 'WEAK';
}
```
